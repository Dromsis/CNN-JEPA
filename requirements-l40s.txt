# Dependencies for the YOLO26-JEPA pretrain on Ada GPUs (L40S / sm_89).
#
# torch / torchvision are NOT listed here on purpose: they come from the base image
# (pytorch/pytorch:2.0.1-cuda11.8-cudnn8-devel -> torch 2.0.1 + cu11.8, runs on L40S).
# Listing the old torch==1.10 pins from requirements.txt would downgrade CUDA and break
# the GPU. Only the framework layer on top of torch is pinned here.

# Lightning stays on 1.9.x: last series that keeps the 1.8-style APIs this codebase uses
# (val_dataloaders[0], precision="bf16", lightning_fabric rank helpers). 2.x would need code
# changes. 1.9.5 officially supports torch 2.0.
pytorch-lightning==1.9.5
torchmetrics==0.11.4

# Self-supervised building blocks (losses, transforms, schedulers, LARS, knn benchmark).
lightly==1.4.21

# timm 0.9.x: still exposes timm.models.registry and timm.models.layers used by the models.
# timm 1.x moved/removed these import paths.
timm==0.9.2

# Config / experiment plumbing.
hydra-core==1.3.2
omegaconf==2.3.0
wandb

# Data + misc utilities actually imported by the pretrain code.
h5py
matplotlib
tqdm
# torch 2.0.1 is NOT compatible with numpy 2.x -> keep numpy on the 1.x line.
numpy==1.26.4

# The YOLO26-SEA backbone is loaded through the vendored `ultralytics` package, which pulls
# these at import time (cv2 etc.). headless OpenCV = no GUI/X11 deps, right for a container.
opencv-python-headless
scipy
psutil
requests
pandas
py-cpuinfo
